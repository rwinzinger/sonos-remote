# Spickzettel — verifizierte Befehle für dieses Board
# (Ausführliche Begründungen stehen in ../CLAUDE.md. Bei Widerspruch gilt CLAUDE.md.)

# --- FQBN ---------------------------------------------------------------------------
# huge_app statt default: LVGL-Builds passen nicht in die 1,3 MB der Default-Partition.
# CDCOnBoot=cdc: sonst landet Serial auf UART0 (FPC-Stecker) und man sieht nichts.
FQBN="esp32:esp32:esp32s3:PSRAM=opi,FlashSize=16M,PartitionScheme=huge_app,CDCOnBoot=cdc"

# ESP32-Core 2.0.17 ist Pflicht — Arduino_GFX 1.3.1 baut nicht gegen Core 3.x.

# --- Port finden --------------------------------------------------------------------
# Der Portname ändert sich, sobald das Board an einer anderen USB-Buchse hängt
# (gesehen: usbmodem21401 -> usbmodem21201 -> usbmodem11101). Nie hart eintragen.
arduino-cli board list
PORT=$(ls /dev/cu.usbmodem* | head -1)

# --- Kompilieren --------------------------------------------------------------------
arduino-cli compile --fqbn "$FQBN" .

# --- Flashen ------------------------------------------------------------------------
# ACHTUNG: "arduino-cli upload" scheitert auf diesem Board am Flasher-Stub
# ("No serial data received" / "Unable to verify flash chip connection").
# Zuverlässig ist esptool mit --no-stub bei 115200:
BD=$(ls -td ~/Library/Caches/arduino/sketches/*/ | xargs -I{} sh -c 'test -f "{}sonos-remote.ino.bin" && echo {}' | head -1)
BOOT_APP0=~/Library/Arduino15/packages/esp32/hardware/esp32/2.0.17/tools/partitions/boot_app0.bin
ESPTOOL=~/Library/Arduino15/packages/esp32/tools/esptool_py/4.5.1/esptool

"$ESPTOOL" --chip esp32s3 --port "$PORT" --baud 115200 --no-stub \
  --before default_reset --after hard_reset \
  write_flash -z --flash_mode keep --flash_freq keep --flash_size keep \
  0x0     "$BD/sonos-remote.ino.bootloader.bin" \
  0x8000  "$BD/sonos-remote.ino.partitions.bin" \
  0xe000  "$BOOT_APP0" \
  0x10000 "$BD/sonos-remote.ino.bin"

# --flash_mode/freq/size "keep" ist wichtig: erzwingt man dio statt des gebauten qio,
# bootet das Board mit OPI-PSRAM nicht mehr.

# --- Serial lesen -------------------------------------------------------------------
# "arduino-cli monitor" liefert NICHTS, wenn seine Ausgabe in eine Pipe/Datei geht —
# es puffert, und beim Abbrechen ist der Puffer weg. Stattdessen:
stty -f "$PORT" 115200 raw -echo clocal
timeout 30 cat "$PORT" > /tmp/serial.log ; cat /tmp/serial.log

# Das Öffnen von /dev/cu.* löst KEINEN Reset aus. Was nur einmal in setup() gedruckt
# wird, verpasst man dadurch fast immer — deshalb loggt die Firmware zyklisch.

# --- Reset ohne neu zu flashen ------------------------------------------------------
"$ESPTOOL" --port "$PORT" --no-stub --after hard-reset read-mac
# Dabei darf kein "cat" auf dem Port hängen, sonst passiert der Reset nicht.

# --- Sonderfall Display-Demo --------------------------------------------------------
# diagnostics/display_baseline linkt ~6,4 MB (40 MB Bilder in der UI-Library) und passt
# in KEIN Standard-Schema. Dafür liegt dort eine eigene partitions.csv (10 MB App).
# Zusätzlich nötig, weil die Größenprüfung sonst abbricht:
#   arduino-cli compile --fqbn "$FQBN" --build-property upload.maximum_size=10485760 .

# --- Bootloop-Verdacht --------------------------------------------------------------
# Kein Serial-Output heißt nicht automatisch "abgestürzt": Der USB-Serial-JTAG-Port
# existiert auch ohne laufende Firmware. Reset-Grund und Boot-Zähler stehen im Banner.
