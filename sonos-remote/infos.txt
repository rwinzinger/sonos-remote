# Kompilieren
arduino-cli compile --fqbn "esp32:esp32:esp32s3:PSRAM=opi,FlashSize=16M,CDCOnBoot=cdc,PartitionScheme=default" .

# Flashen
arduino-cli upload -p /dev/cu.usbmodem21401 --fqbn "esp32:esp32:esp32s3:PSRAM=opi,FlashSize=16M,CDCOnBoot=cdc,PartitionScheme=default" .

# Serial-Monitor (der Feedback-Kanal)
arduino-cli monitor -p /dev/cu.usbmodem21401 -c baudrate=115200
